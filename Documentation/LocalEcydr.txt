
// I/O Type Control //

//--------------------------------------------------------------------------------------------//
// Move Forward
//--------------------------------------------------------------------------------------------//
	if(gWs_EcdrFwd_Start==1)
	{
		gWs_EcdrFwd_Start = 0;
		call spInitEcylinder();
		gWs_EcdrFwd_Seq = 1;
		gWs_EcdrFwd_Ena = 1;
	}

	if(gWs_EcdrFwd_Abort==1)
	{
		gWs_EcdrFwd_Abort = 0;
		gWs_EcdrFwd_Seq = 0;
		gWs_EcdrFwd_Ena = 0;
	}

	if(gWs_EcdrFwd_Ena==1)
	{
		switch(gWs_EcdrFwd_Seq)
		{
			case 1:
				gWs_EcdrFwd_Buf = gWs_EcdrFwd_Seq;
				if(pEcat_EcldrDi_Estop==1 || pEcat_EcldrDi_Alarm==1)
				{
					if(gWs_EcdrFwd_Try < cEcdrTry_Num)
					{
						pEcat_EcldrDo_Reset = 0;
						gWs_EcdrFwd_Tmr = Sys.Time + 0.2;
						gWs_EcdrFwd_Seq = 2;
					}
					else
					{
						gWs_EcdrFwd_Sta = cEcdrSta_Err;
						gWs_EcdrFwd_Seq = 0;
						gWs_EcdrFwd_Ena = 0;
					}
				}
				else
				{
					gWs_EcdrFwd_Sta = cEcdrSta_Ing;
					gWs_EcdrFwd_Seq = 10;
				}
			break;

			case 2:
				gWs_EcdrFwd_Buf = gWs_EcdrFwd_Seq;
				if(Sys.Time > gWs_EcdrFwd_Tmr)
				{
					pEcat_EcldrDo_Reset = 1;
					gWs_EcdrFwd_Tmr = Sys.Time + 0.2;
					gWs_EcdrFwd_Seq = 3;
				}
			break;

			case 3:
				gWs_EcdrFwd_Buf = gWs_EcdrFwd_Seq;
				if(Sys.Time > gWs_EcdrFwd_Tmr)
				{
					gWs_EcdrFwd_Try++;
					gWs_EcdrFwd_Seq = 1;
				}
			break;
		//----------------------------------------//
			case 10:
				gWs_EcdrFwd_Buf = gWs_EcdrFwd_Seq;
				pEcat_EcldrDo_Svon = 1;
				gWs_EcdrFwd_Tmr = Sys.Time + 3;
				gWs_EcdrFwd_Seq = 11;
			break;

			case 11:
				if(pEcat_EcldrDi_Svre==1)
				{
					pEcat_EcldrDo_In0 = 1;
					pEcat_EcldrDo_In1 = 0;
					pEcat_EcldrDo_Drive = 0;
					gWs_EcdrFwd_Tmr = Sys.Time + 0.2;
					gWs_EcdrFwd_Seq = 12;
				}

				if(Sys.Time > gWs_EcdrFwd_Tmr)
				{
					gWs_EcdrFwd_Sta = cEcdrSta_Err;
					gWs_EcdrFwd_Seq = 0;
					gWs_EcdrFwd_Ena = 0;
				}
			break;

			case 12:
				gWs_EcdrFwd_Buf = gWs_EcdrFwd_Seq;
				if(Sys.Time > gWs_EcdrFwd_Tmr)
				{
					pEcat_EcldrDo_Drive = 1;
					//gWs_EcdrFwd_Tmr = Sys.Time + 0.5;
					gWs_EcdrFwd_Seq = 13;
				}
			break;

			case 13:
				gWs_EcdrFwd_Buf = gWs_EcdrFwd_Seq;
				//if(Sys.Time > gWs_EcdrFwd_Tmr)
				if(pEcat_EcldrDi_Inp==0)
				{
					gWs_EcdrFwd_Tmr = Sys.Time + 10;
					gWs_EcdrFwd_Seq = 14;
				}
			break;

			case 14:
				gWs_EcdrFwd_Buf = gWs_EcdrFwd_Seq;
				if(pEcat_EcldrDi_Seton==1 && pEcat_EcldrDi_Inp==1)
				{
					gWs_EcdrFwd_Sta = cEcdrSta_Fin;
					gWs_EcdrFwd_Seq = 0;
					gWs_EcdrFwd_Ena = 0;
				}

				if(Sys.Time > gWs_EcdrFwd_Tmr)
				{
					gWs_EcdrFwd_Sta = cEcdrSta_Err;
					gWs_EcdrFwd_Seq = 0;
					gWs_EcdrFwd_Ena = 0;
				}
			break;

			default:
				gWs_EcdrFwd_Seq = 0;
				gWs_EcdrFwd_Ena = 0;
			break;
		}
	}

//--------------------------------------------------------------------------------------------//
// Move Backward
//--------------------------------------------------------------------------------------------//
	if(gWs_EcdrBwd_Start==1)
	{
		gWs_EcdrBwd_Start = 0;
		call spInitEcylinder();
		gWs_EcdrBwd_Seq = 1;
		gWs_EcdrBwd_Ena = 1;
	}

	if(gWs_EcdrBwd_Abort==1)
	{
		gWs_EcdrBwd_Abort = 0;
		gWs_EcdrBwd_Seq = 0;
		gWs_EcdrBwd_Ena = 0;
	}

	if(gWs_EcdrBwd_Ena==1)
	{
		switch(gWs_EcdrBwd_Seq)
		{
			case 1:
				gWs_EcdrBwd_Buf = gWs_EcdrBwd_Seq;
				if(pEcat_EcldrDi_Estop==1 || pEcat_EcldrDi_Alarm==1)
				{
					if(gWs_EcdrBwd_Try < cEcdrTry_Num)
					{
						pEcat_EcldrDo_Reset = 0;
						gWs_EcdrBwd_Tmr = Sys.Time + 0.2;
						gWs_EcdrBwd_Seq = 2;
					}
					else
					{
						gWs_EcdrBwd_Sta = cEcdrSta_Err;
						gWs_EcdrBwd_Seq = 0;
						gWs_EcdrBwd_Ena = 0;
					}
				}
				else
				{
					gWs_EcdrBwd_Sta = cEcdrSta_Ing;
					gWs_EcdrBwd_Seq = 10;
				}
			break;

			case 2:
				gWs_EcdrBwd_Buf = gWs_EcdrBwd_Seq;
				if(Sys.Time > gWs_EcdrBwd_Tmr)
				{
					pEcat_EcldrDo_Reset = 1;
					gWs_EcdrBwd_Tmr = Sys.Time + 0.2;
					gWs_EcdrBwd_Seq = 3;
				}
			break;

			case 3:
				gWs_EcdrBwd_Buf = gWs_EcdrBwd_Seq;
				if(Sys.Time > gWs_EcdrBwd_Tmr)
				{
					gWs_EcdrBwd_Try++;
					gWs_EcdrBwd_Seq = 1;
				}
			break;
		//----------------------------------------//
			case 10:
				gWs_EcdrBwd_Buf = gWs_EcdrBwd_Seq;
				pEcat_EcldrDo_Svon = 1;
				gWs_EcdrBwd_Tmr = Sys.Time + 3;
				gWs_EcdrBwd_Seq = 11;
			break;

			case 11:
				if(pEcat_EcldrDi_Svre==1)
				{
					pEcat_EcldrDo_In0 = 0;
					pEcat_EcldrDo_In1 = 1;
					pEcat_EcldrDo_Drive = 0;
					gWs_EcdrBwd_Tmr = Sys.Time + 0.2;
					gWs_EcdrBwd_Seq = 12;
				}

				if(Sys.Time > gWs_EcdrBwd_Tmr)
				{
					gWs_EcdrBwd_Sta = cEcdrSta_Err;
					gWs_EcdrBwd_Seq = 0;
					gWs_EcdrBwd_Ena = 0;
				}
			break;

			case 12:
				gWs_EcdrBwd_Buf = gWs_EcdrBwd_Seq;
				if(Sys.Time > gWs_EcdrBwd_Tmr)
				{
					pEcat_EcldrDo_Drive = 1;
					//gWs_EcdrBwd_Tmr = Sys.Time + 0.5;
					gWs_EcdrBwd_Seq = 13;
				}
			break;

			case 13:
				gWs_EcdrBwd_Buf = gWs_EcdrBwd_Seq;
				//if(Sys.Time > gWs_EcdrBwd_Tmr)
				if(pEcat_EcldrDi_Inp==0)
				{
					gWs_EcdrBwd_Tmr = Sys.Time + 10;
					gWs_EcdrBwd_Seq = 14;
				}
			break;

			case 14:
				gWs_EcdrBwd_Buf = gWs_EcdrBwd_Seq;
				if(pEcat_EcldrDi_Seton==1 && pEcat_EcldrDi_Inp==1)
				{
					gWs_EcdrBwd_Sta = cEcdrSta_Fin;
					gWs_EcdrBwd_Seq = 0;
					gWs_EcdrBwd_Ena = 0;
				}

				if(Sys.Time > gWs_EcdrBwd_Tmr)
				{
					gWs_EcdrBwd_Sta = cEcdrSta_Err;
					gWs_EcdrBwd_Seq = 0;
					gWs_EcdrBwd_Ena = 0;
				}
			break;

			default:
				gWs_EcdrBwd_Seq = 0;
				gWs_EcdrBwd_Ena = 0;
			break;
		}
	}
